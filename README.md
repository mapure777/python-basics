my_dict = {"apple": "a fruit", "book": "a set of written or printed pages"}
print(my_dict, '/n')

#adding a new key value pair

my_dict ["car"] = " a vehicle"
print(my_dict,'/n')

#deleting an item

my_dict["apple"] = "a sweet fruit"
print(my_dict, '/n')

#deleting an item

del my_dict["book"]
print(my_dict, '/n')

#removing an item

my_dict.pop("car")
print(my_dict, '/n')
